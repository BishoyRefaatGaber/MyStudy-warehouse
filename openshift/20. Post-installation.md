## Identity providers
#### 1. htpasswd
- Create the htpasswd file and secret.
```bash
sudo dnf -y install httpd-tools

htpasswd -c -B -b </path/to/users.htpasswd> <user_name> <password>

oc create secret generic htpass-secret --from-file=htpasswd=<path_to_users.htpasswd> -n openshift-config
```
- Create a custom CR
```yaml
apiVersion: config.openshift.io/v1
kind: OAuth
metadata:
  name: cluster
spec:
  identityProviders:
  - name: my_htpasswd_provider 
    mappingMethod: claim 
    type: HTPasswd
    htpasswd:
      fileData:
        name: htpass-secret 
```
```bash
oc apply -f </path/to/CR>
```


#### 2. LDAP

- LDAP URL: ldap://host:port/basedn?attribute?scope?filter
	- attribute: uid, sAMAccountName
-  you must define an OpenShift Container Platform `Secret` object that contains the `bindPassword` field.
```bash
oc create secret generic ldap-secret --from-literal=bindPassword=<secret> -n openshift-config
```
- Define an OpenShift Container Platform `ConfigMap` object containing the certificate authority by using the following command. The certificate authority must be stored in the `ca.crt` key of the `ConfigMap` object.
```bash
oc create configmap ca-config-map --from-file=ca.crt=/path/to/ca -n openshift-config
```
- Create the LDAP CR
```yaml
apiVersion: config.openshift.io/v1
kind: OAuth
metadata:
  name: cluster
spec:
  identityProviders:
  - name: ldapidp 
    mappingMethod: claim 
    type: LDAP
    ldap:
      attributes:
        id: 
        - dn
        name: 
        - cn
        preferredUsername: 
        - sAMAccountName
        - uid
      bindDN: "" 
      bindPassword: 
        name: ldap-secret
      ca: 
        name: ca-config-map
      insecure: false 
      url: "ldaps://ldaps.example.com/ou=users,dc=acme,dc=com?uid" 
```
- Apply the CR
```bash
oc apply -f </path/to/CR>
```

## Create Infra nodes

- Label infra nodes
```bash
oc label node <node-name> node-role.kubernetes.io/infra=""
oc edit scheduler cluster
	defaultNodeSelector: node-role.kubernetes.io/infra=""
```
- Create MachineConfigPool
```yaml
apiVersion: machineconfiguration.openshift.io/v1
kind: MachineConfigPool
metadata:
  name: infra
spec:
  machineConfigSelector:
    matchExpressions:
      - {key: machineconfiguration.openshift.io/role, operator: In, values: [worker,infra]}
  nodeSelector:
    matchLabels:
      node-role.kubernetes.io/infra: ""
```
```bash
oc apply -f <filename>
```
- Taint the infra nodes
```bash 
oc adm taint nodes node1 node-role.kubernetes.io/infra=reserved:NoSchedule
```
- Move router to infra
```bash 
oc edit ingresscontroller default -n openshift-ingress-operator

spec:
  nodePlacement:
    nodeSelector:
      matchLabels:
        node-role.kubernetes.io/infra: ""
    tolerations:
    - effect: NoSchedule
      key: node-role.kubernetes.io/infra
      value: reserved
      
# Verify      
oc get pod -n openshift-ingress -o wide      
```
- Move the default registry
```bash
oc edit configs.imageregistry.operator.openshift.io/cluster

spec:
  logLevel: Normal
  managementState: Managed

  tolerations:
  - effect: NoSchedule
    key: node-role.kubernetes.io/infra
    value: reserved
    
# Verify    
oc get pods -o wide -n openshift-image-registry    
```

## 3. NTP
```bash
sudo dnf -y install butane



vim 99-master-chrony.bu


variant: openshift
version: 4.20.0
metadata:
  name: 99-master-chrony
  labels:
    machineconfiguration.openshift.io/role: master
storage:
  files:
    - path: /etc/chrony.conf
      mode: 0644
      overwrite: true
      contents:
        inline: |
          # Internal or preferred NTP servers
          server ntp1.internal.example.com iburst
          server ntp2.internal.example.com iburst
          
          driftfile /var/lib/chrony/drift
          makestep 1.0 3
          rtcsync
          logdir /var/log/chrony



butane 99-master-chrony.bu -o 99-master-chrony.yaml


add this at the end of 99-master-chrony.yaml  for time zone 

    systemd:
      units:
        - contents: |
            [Unit]
            Description=set timezone
            After=network-online.target
            [Service]
            Type=oneshot
            ExecStart=timedatectl set-timezone Africa/Cairo
            [Install]
            WantedBy=multi-user.target
          enabled: true
          name: custom-timezone.service

```

## 4. Securing \*.apps
- Create a config map that includes only the root CA certificate that is used to sign the wildcard certificate
```bash
oc create configmap custom-ca \ --from-file=ca-bundle.crt=</path/to/example-ca.crt> \ -n openshift-config
```
- Update the cluster-wide proxy configuration with the newly created config map:
```bash
oc patch proxy/cluster \
     --type=merge \
     --patch='{"spec":{"trustedCA":{"name":"custom-ca"}}}'
```
- Create a secret that contains the wildcard certificate chain and key
```bash
oc create secret tls <secret> \
     --cert=</path/to/cert.crt> \
     --key=</path/to/cert.key> \
     -n openshift-ingress
```
- Update the Ingress Controller configuration with the newly created secret
```bash
oc patch ingresscontroller.operator default \
     --type=merge -p \
     '{"spec":{"defaultCertificate": {"name": "<secret>"}}}' \
     -n openshift-ingress-operator
```

## 5. Log Forwarder

- Install the Red Hat OpenShift Logging Operator in the `openshift-logging` namespace.
- Install the Loki Operator and a `LokiStack` instance named `logging-loki` exists in the `openshift-logging` namespace if you are forwarding logs to LokiStack.
- Create a Service Account 
```bash
oc create sa collector -n openshift-logging
```
- Bind the appropriate cluster roles to the service account
```bash
# Grant the service account permission to collect application logs
oc adm policy add-cluster-role-to-user collect-application-logs system:serviceaccount:openshift-logging:collector

# Grant the service account permission to collect infrastructure logs
oc adm policy add-cluster-role-to-user collect-infrastructure-logs system:serviceaccount:openshift-logging:collector

# Grant the service account permission to collect audit logs
oc adm policy add-cluster-role-to-user collect-audit-logs system:serviceaccount:openshift-logging:collector
```
- Create a `ClusterLogForwarder` custom resource (CR) to define your log forwarding pipeline.
```yaml
# example to forward application and infrastructure logs to an in-cluster `LokiStack`

apiVersion: observability.openshift.io/v1
kind: ClusterLogForwarder
metadata:
  name: instance
  namespace: openshift-logging
spec:
  serviceAccount:
    name: collector
  outputs:
  - name: default-lokistack
    type: lokiStack
    lokiStack:
      authentication:
        token:
          from: serviceAccount
      target:
        name: logging-loki
        namespace: openshift-logging
    tls:
      ca:
        configMapName: logging-loki-gateway-ca-bundle
        key: service-ca.crt
  pipelines:
  - name: default-logstore
    inputRefs:
    - application
    - infrastructure
    outputRefs:
    - default-lokistack
```
- Apply the CR
```bash
oc apply -f <filename>.yaml
```
- Verify
```bash
oc get clusterlogforwarder instance -n openshift-logging \
  -o jsonpath='{.status.conditions[?(@.type=="Ready")].status}'
```
- The output should show `True`. If the output shows `False`, check the `status.conditions` for error messages
```bash
oc get clusterlogforwarder instance -n openshift-logging \
  -o jsonpath='{.status.conditions}' | python3 -m json.tool
```
